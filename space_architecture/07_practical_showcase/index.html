驾驶脚本 // AssemblyScript · 26 行
// 登机推杆：W 加推力；推力托起后向前掠出悬崖
const held = !ctx.driver.isNull && ctx.input.down("KeyW");
spool = held ? min(1, spool + dt*0.55) : max(0, spool - dt*0.4);
let lift = max(0, spool - 0.5) * 12.0;
h = h + lift * dt;
if (h > 1.2) v = min(22.0, v + spool*9.0*dt);
self.setLocalPosition([x0, y0 + h, z0 + d]);
